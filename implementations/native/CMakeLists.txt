cmake_minimum_required(VERSION 3.20)
project(flir_agc_native VERSION 1.0 LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
option(AGC_AVX2 "Compile custom loops for AVX2-capable CPUs" ON)
add_library(flir_agc SHARED cpp/agc.cpp)
target_compile_features(flir_agc PRIVATE cxx_std_17)
target_include_directories(flir_agc PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(flir_agc PRIVATE ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(flir_agc PRIVATE /O2 /Oi /fp:precise /W4)
  if(AGC_AVX2)
    target_compile_options(flir_agc PRIVATE /arch:AVX2)
  endif()
else()
  target_compile_options(flir_agc PRIVATE -O3 -ffp-contract=off -Wall -Wextra)
  if(AGC_AVX2)
    target_compile_options(flir_agc PRIVATE -mavx2)
  endif()
endif()
add_executable(agc_example cpp/example.cpp)
target_compile_features(agc_example PRIVATE cxx_std_17)
target_include_directories(agc_example PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(agc_example PRIVATE flir_agc ${OpenCV_LIBS})
