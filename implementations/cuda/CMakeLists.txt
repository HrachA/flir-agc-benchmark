cmake_minimum_required(VERSION 3.24)
project(flir_agc_cuda VERSION 1.0 LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(flir_agc_cuda SHARED cuda/agc.cu)
target_compile_features(flir_agc_cuda PRIVATE cxx_std_17 cuda_std_17)
set(AGC_CUDA_ARCHITECTURES "89" CACHE STRING "CUDA target architectures; 89 for RTX 4070 Ti SUPER")
set_target_properties(flir_agc_cuda PROPERTIES CUDA_ARCHITECTURES "${AGC_CUDA_ARCHITECTURES}" CUDA_SEPARABLE_COMPILATION OFF)
target_compile_options(flir_agc_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;--prec-div=true;--prec-sqrt=true;-lineinfo>)
target_link_libraries(flir_agc_cuda PRIVATE CUDA::cudart)
